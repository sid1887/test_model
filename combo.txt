# Multi-stage build for production optimization
FROM python:3.11-slim as base

# Build-time environment variables for timeout handling
ENV PIP_DEFAULT_TIMEOUT=2000
ENV PIP_TIMEOUT=2000
ENV PIP_RETRIES=10
ENV PYTHONUNBUFFERED=1
ENV PYTHONDONTWRITEBYTECODE=1
ENV DEBIAN_FRONTEND=noninteractive

# Set working directory
WORKDIR /app

# Stage 1: Dependencies and build tools
FROM base as builder

# Install system dependencies with timeout handling
RUN apt-get update && apt-get install -y \
    --no-install-recommends \
    gcc \
    g++ \
    libpq-dev \
    tesseract-ocr \
    tesseract-ocr-eng \
    libglib2.0-0 \
    libsm6 \
    libxext6 \
    libxrender-dev \
    libgomp1 \
    libgl1-mesa-glx \
    libgtk-3-0 \
    libavcodec-dev \
    libavformat-dev \
    libswscale-dev \
    libv4l-dev \
    libatlas-base-dev \
    gfortran \
    libjpeg-dev \
    libpng-dev \
    libtiff-dev \
    curl \
    git \
    wget \
    build-essential \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# Create pip configuration for enhanced timeout handling
RUN mkdir -p ~/.pip && \
    echo "[global]" > ~/.pip/pip.conf && \
    echo "trusted-host = pypi.org" >> ~/.pip/pip.conf && \
    echo "               files.pythonhosted.org" >> ~/.pip/pip.conf && \
    echo "               pypi.python.org" >> ~/.pip/pip.conf && \
    echo "timeout = 2000" >> ~/.pip/pip.conf && \
    echo "retries = 10" >> ~/.pip/pip.conf && \
    echo "break-system-packages = true" >> ~/.pip/pip.conf

# Copy requirements first for better caching
COPY requirements.txt .

# Install Python dependencies with comprehensive timeout settings
RUN pip install --no-cache-dir \
    --timeout 2000 \
    --retries 10 \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    --index-url https://pypi.org/simple \
    --upgrade pip setuptools wheel && \
    pip install --no-cache-dir \
    --timeout 2000 \
    --retries 10 \
    --trusted-host pypi.org \
    --trusted-host files.pythonhosted.org \
    --trusted-host pypi.python.org \
    -r requirements.txt

# Install Playwright browsers with timeout handling
RUN playwright install chromium --force && \
    playwright install-deps chromium

# Stage 2: Production runtime
FROM base as production

# Install only runtime dependencies
RUN apt-get update && apt-get install -y \
    --no-install-recommends \
    tesseract-ocr \
    tesseract-ocr-eng \
    libglib2.0-0 \
    libsm6 \
    libxext6 \
    libxrender1 \
    libgomp1 \
    libgl1-mesa-glx \
    libgtk-3-0 \
    libavcodec58 \
    libavformat58 \
    libswscale5 \
    libv4l-0 \
    libatlas3-base \
    libjpeg62-turbo \
    libpng16-16 \
    libtiff5 \
    curl \
    wget \
    procps \
    && rm -rf /var/lib/apt/lists/* \
    && apt-get clean

# Create non-root user for security
RUN groupadd -r appuser && useradd -r -g appuser -u 1001 appuser && \
    mkdir -p /app /uploads /models /app/static /app/.cache && \
    chown -R appuser:appuser /app /uploads /models

# Copy Python packages from builder stage
COPY --from=builder /usr/local/lib/python3.11/site-packages /usr/local/lib/python3.11/site-packages
COPY --from=builder /usr/local/bin /usr/local/bin
COPY --from=builder /root/.cache/ms-playwright /home/appuser/.cache/ms-playwright

# Copy application code
COPY --chown=appuser:appuser . .

# Ensure main.py is present
COPY --chown=appuser:appuser main.py /app/main.py

# Set proper ownership for Playwright
RUN chown -R appuser:appuser /home/appuser/.cache

# Switch to non-root user
USER appuser

# Create necessary directories with proper permissions
RUN mkdir -p /uploads /models /app/static /app/.cache

# Set environment variables for production
ENV PYTHONPATH=/app
ENV PATH="/home/appuser/.local/bin:$PATH"

# Expose port
EXPOSE 8000

# Health check
HEALTHCHECK --interval=30s --timeout=20s --start-period=120s --retries=5 \
    CMD curl -f http://localhost:8000/api/v1/health || exit 1

# Default command with proper signal handling
CMD ["uvicorn", "main:app", "--host", "0.0.0.0", "--port", "8000", "--workers", "4"]
